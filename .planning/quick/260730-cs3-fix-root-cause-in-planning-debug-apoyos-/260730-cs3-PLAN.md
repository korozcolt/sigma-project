Only the signature line and the one `$voter->campaign_id` line change — every other line of the method body stays identical, in the same order:

```php
/**
 * Automated/headless cascade (D-03): permanent lookup -> live -> national snapshot.
 * Never blocks (LIVE-03). $voter may be null (e.g. VoterValidationService checking
 * census existence before a Voter record exists) — persist() already treats a null
 * $voter as a pure pass-through. $campaignId is used only for the permanent-lookup
 * write's informational campaign_id column when $voter is null.
 */
public function resolveAutomated(string $cedula, ?Voter $voter = null, string $resolvedVia = 'reconciliation', ?int $campaignId = null): ?PollingPlaceResolutionResult
{
    if ($fromPermanentLookup = $this->resolveFromPermanentLookup($cedula)) {
        return $this->persist($voter, $fromPermanentLookup, isExplicitOverride: false, resolvedVia: $resolvedVia);
    }

    foreach ($this->liveAdapters as $adapter) {
        if ($fields = $this->attemptLiveAutomated($adapter, $cedula)) {
            $this->persistPermanentLookup($cedula, $fields, $voter?->campaign_id ?? $campaignId);

            $result = new PollingPlaceResolutionResult(
                source: PollingPlaceSource::LIVE,
                fields: $fields,
                pollingPlaceId: $this->resolveOrCreatePollingPlace($fields)?->id,
                tableNumber: ltrim($fields['mesa_numero'] ?? '', '0') ?: null,
            );

            return $this->persist($voter, $result, isExplicitOverride: false, resolvedVia: $resolvedVia);
        }
    }

    if ($fromSnapshot = $this->resolveFromNationalSnapshot($cedula)) {
        return $this->persist($voter, $fromSnapshot, isExplicitOverride: false, resolvedVia: $resolvedVia);
    }

    return null;
}
```

2. Replace `app/Services/VoterValidationService.php` in full:

```php
<?php

namespace App\Services;

use App\Enums\PollingPlaceSource;
use App\Enums\VoterStatus;
use App\Models\CensusRecord;
use App\Models\ValidationHistory;
use App\Models\Voter;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\DB;

class VoterValidationService
{
    /**
     * Validate a voter against the census by delegating to the SAME cascade
     * PollingPlaceResolver::resolveAutomated() already uses for polling-place resolution:
     * permanent registraduria_lookups cache -> live Registraduría adapters (wsp/infovotantes)
     * -> national census snapshot (national_census_records). Falls back to
     * national_identity_records (cross-instance cedula->nombre catalog) as a final
     * existence-only check when the resolver's cascade is fully exhausted.
     *
     * Replaces the old orphaned per-campaign CensusRecord lookup — see
     * .planning/debug/apoyos-mass-rejected-census-betha.md. CensusRecord/CensusImporter
     * are no longer the source of truth for this check (kept on disk unused — see
     * getCensusInfo() below, which is the only remaining reader).
     *
     * BEHAVIOR CHANGE: registraduria_lookups/national_census_records/national_identity_records
     * are cross-instance/global catalogs (not per-campaign), so this check is intentionally
     * no longer campaign-isolated the way the old CensusRecord lookup was.
     *
     * @return array{found: bool, source: ?PollingPlaceSource, confidence: string}
     */
    public function validateAgainstCensus(Voter $voter): array
    {
        $result = app(PollingPlaceResolver::class)->resolveAutomated(
            $voter->document_number,
            $voter,
            resolvedVia: 'census_validation',
        );

        if ($result !== null) {
            return ['found' => true, 'source' => $result->source, 'confidence' => 'high'];
        }

        if (app(IdentityLookupService::class)->findByDocumentNumber($voter->document_number)) {
            return ['found' => true, 'source' => null, 'confidence' => 'medium'];
        }

        return ['found' => false, 'source' => null, 'confidence' => 'none'];
    }

    /**
     * Update voter status based on census validation. REJECTED_CENSUS is now only ever
     * set once the full cascade above (cache -> live -> national snapshot -> national
     * identity) has been exhausted with no match.
     */
    public function updateVoterStatus(Voter $voter, bool $found): Voter
    {
        $previousStatus = $voter->status;

        if ($found) {
            $voter->update([
                'status' => VoterStatus::VERIFIED_CENSUS,
                'census_validated_at' => now(),
            ]);
        } else {
            $voter->update([
                'status' => VoterStatus::REJECTED_CENSUS,
                'notes' => 'No se encontró en el censo electoral',
            ]);
        }

        ValidationHistory::create([
            'voter_id' => $voter->id,
            'previous_status' => $previousStatus,
            'new_status' => $voter->status,
            'validated_by' => Auth::id(),
            'validation_type' => 'census',
            'notes' => $found
                ? 'Validado exitosamente contra el censo electoral'
                : 'No se encontró en el censo electoral',
        ]);

        return $voter->fresh();
    }

    /**
     * @return array{voter: Voter, found: bool, source: ?PollingPlaceSource}
     */
    public function validateAndUpdate(Voter $voter): array
    {
        $result = $this->validateAgainstCensus($voter);

        $updatedVoter = $this->updateVoterStatus($voter, $result['found']);

        return [
            'voter' => $updatedVoter,
            'found' => $result['found'],
            'source' => $result['source'],
        ];
    }

    /**
     * @return array{validated: int, found: int, rejected: int}
     */
    public function validatePendingVoters(int $campaignId): array
    {
        $pendingVoters = Voter::where('campaign_id', $campaignId)
            ->where('status', VoterStatus::PENDING_REVIEW->value)
            ->get();

        $validated = 0;
        $found = 0;
        $rejected = 0;

        DB::beginTransaction();

        try {
            foreach ($pendingVoters as $voter) {
                $result = $this->validateAgainstCensus($voter);

                $this->updateVoterStatus($voter, $result['found']);

                $validated++;

                if ($result['found']) {
                    $found++;
                } else {
                    $rejected++;
                }
            }

            DB::commit();

            return [
                'validated' => $validated,
                'found' => $found,
                'rejected' => $rejected,
            ];
        } catch (\Exception $e) {
            DB::rollBack();

            throw $e;
        }
    }

    /**
     * Checks census existence via the same automated cascade used for validation, without
     * requiring a persisted Voter (used before a Voter record exists — e.g.
     * register-voter.blade.php's/create-leader.blade.php's blur handlers and pre-save
     * checks). $campaignId is kept for call-site signature compatibility and passed
     * through only as informational metadata on a fresh live hit.
     */
    public function documentExistsInCensus(int $campaignId, string $documentNumber): bool
    {
        $result = app(PollingPlaceResolver::class)->resolveAutomated(
            $documentNumber,
            voter: null,
            resolvedVia: 'census_validation',
            campaignId: $campaignId,
        );

        if ($result !== null) {
            return true;
        }

        return app(IdentityLookupService::class)->findByDocumentNumber($documentNumber) !== null;
    }

    /**
     * @deprecated Legacy per-campaign CensusRecord lookup. CensusRecord is NO LONGER the
     * source of truth for census validation (see validateAgainstCensus()/
     * documentExistsInCensus() above) — retained only to surface the incidental rows
     * HasRegistraduriaPolling.php still writes there. Do not use this to gate new logic.
     */
    public function getCensusInfo(Voter $voter): ?CensusRecord
    {
        return CensusRecord::where('campaign_id', $voter->campaign_id)
            ->where('document_number', $voter->document_number)
            ->first();
    }
}
```

```php
<?php

namespace App\Jobs;

use App\Models\CensusRevalidationRun;
use App\Models\Voter;
use App\Services\VoterValidationService;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Queue\Queueable;

class ValidateVoterAgainstCensus implements ShouldQueue
{
    use Queueable;

    public function __construct(
        public Voter $voter,
        public ?int $censusRevalidationRunId = null,
    ) {
        //
    }

    public function handle(VoterValidationService $validationService): void
    {
        $previousStatus = $this->voter->status;

        $validationService->validateAndUpdate($this->voter);

        if ($this->censusRevalidationRunId === null) {
            return;
        }

        $run = CensusRevalidationRun::find($this->censusRevalidationRunId);

        if (! $run) {
            return;
        }

        $changed = $previousStatus !== $this->voter->fresh()->status;

        $run->increment('processed');

        if ($changed) {
            $run->increment('changed');
        }

        $run->refresh();

        if ($run->finished_at === null && $run->processed >= $run->total) {
            $run->update(['finished_at' => now()]);
        }
    }
}

use App\Models\CensusRevalidationRun;
use Filament\Widgets\StatsOverviewWidget;
use Filament\Widgets\StatsOverviewWidget\Stat;

class CensusRevalidationProgressWidget extends StatsOverviewWidget
{
    protected static ?int $sort = 0;

    protected ?string $pollingInterval = '3s';

    protected function getStats(): array
    {
        $run = CensusRevalidationRun::query()->latest('id')->first();

        if (! $run) {
            return [
                Stat::make('Revalidación de Censo', 'Sin ejecuciones registradas')
                    ->color('gray'),
            ];
        }

        if ($run->isInProgress()) {
            return [
                Stat::make('Revalidación de Censo en curso', "Iniciada {$run->started_at->diffForHumans()}")
                    ->description("{$run->processed} / {$run->total} apoyos procesados")
                    ->descriptionIcon('heroicon-m-arrow-path')
                    ->color('info'),
            ];
        }

        return [
            Stat::make('Última revalidación de censo', "Completada {$run->finished_at->diffForHumans()}")
                ->description("{$run->processed} procesados — {$run->changed} cambiaron de estado")
                ->descriptionIcon('heroicon-m-check-circle')
                ->color('success'),
        ];
    }
}

<?php

declare(strict_types=1);

namespace App\Console\Commands;

use App\Enums\VoterStatus;
use App\Models\ValidationHistory;
use App\Models\Voter;
use Illuminate\Console\Command;

class RemediateBethaRejectedCensus extends Command
{
    protected $signature = 'census:remediate-betha-rejections {--apply : Actually persist the change; without this flag the command only previews the affected voters}';

    protected $description = 'One-off data correction for sigma-betha (DB sigma_betha, campaign_id=1 "Alcaldía 2027"): reverts rejected_census voters mis-rejected against the orphaned per-campaign census_records table back to pending_review, writing an audit ValidationHistory row for each. Refuses to run against any database other than sigma_betha.';

    private const int TARGET_CAMPAIGN_ID = 1;

    private const string SAFETY_DATABASE_NAME = 'sigma_betha';

    public function handle(): int
    {
        $connectionName = config('database.default');
        $actualDatabase = config("database.connections.{$connectionName}.database");

        if ($actualDatabase !== self::SAFETY_DATABASE_NAME) {
            $this->error("Rechazado: la base de datos conectada es '{$actualDatabase}', se esperaba '".self::SAFETY_DATABASE_NAME."'. Este comando está limitado exclusivamente a sigma-betha.");

            return self::FAILURE;
        }

        $voters = Voter::query()
            ->where('campaign_id', self::TARGET_CAMPAIGN_ID)
            ->where('status', VoterStatus::REJECTED_CENSUS->value)
            ->get();

        $this->info("Encontrados {$voters->count()} apoyos rejected_census en campaign_id=".self::TARGET_CAMPAIGN_ID." de la base '{$actualDatabase}'.");

        if (! $this->option('apply')) {
            $this->comment('Dry-run (sin --apply): no se modificó ningún registro.');

            return self::SUCCESS;
        }

        $corrected = 0;

        foreach ($voters as $voter) {
            $previousStatus = $voter->status;

            $voter->update([
                'status' => VoterStatus::PENDING_REVIEW,
                'notes' => 'Corrección de datos (quick task 260730-cs3): revertido desde rejected_census. El rechazo original comparó contra la tabla census_records huérfana de esta campaña (45 filas incidentales), no contra el censo/Registraduría real.',
            ]);

            ValidationHistory::create([
                'voter_id' => $voter->id,
                'previous_status' => $previousStatus,
                'new_status' => VoterStatus::PENDING_REVIEW,
                'validated_by' => null,
                'validation_type' => 'data_correction',
                'notes' => 'Remediación de datos (quick task 260730-cs3): rejected_census -> pending_review. Root cause: validación contra census_records huérfana en vez del cascade real (registraduria_lookups / live / censo nacional / identidad nacional).',
            ]);

            $corrected++;
        }

        $this->info("Corregidos {$corrected} apoyos: rejected_census -> pending_review.");

        return self::SUCCESS;
    }
}
```
